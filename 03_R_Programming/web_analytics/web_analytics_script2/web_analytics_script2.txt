# 4. DATA CLEANING AND FEATURE PREPARATION
# ===================================================================

# Remove duplicate records
clean_data <- raw_data %>%
distinct()

# Main cleaning rule:
# Remove rows with Visits = 0 because visits represent valid sessions.
# Rows with zero visits cannot be reliably used in web activity modelling.
clean_data <- clean_data %>%
filter(Visits > 0)

# Convert categorical variables to factors
clean_data <- clean_data %>%
mutate(
  Continent = as.factor(Continent),
  Sourcegroup = as.factor(Sourcegroup)
)

# Rename columns for clearer coding
clean_data <- clean_data %>%
rename(
  SourceGroup = Sourcegroup,
  TimeOnPage = Timeinpage,
  UniquePageviews = Uniquepageviews
)

# Create supporting per-visit metrics
clean_data <- clean_data %>%
mutate(
  AvgTimeOnPage = TimeOnPage / Visits,
  ViewsPerVisit = UniquePageviews / Visits,
  BouncePerVisit = Bounces / Visits,
  ExitPerVisit = Exits / Visits
)

# Decide how to treat Bounces:
# If Bounces ranges from 0 to 100, it may already be a percentage.
# If Bounces is greater than 100, it is likely a count.
max_bounces <- max(clean_data$Bounces, na.rm = TRUE)

if (max_bounces <= 100) {
clean_data <- clean_data %>%
  mutate(BounceMetric = Bounces)
bounce_metric_note <- "Bounces appears to be a percentage/proportion-style variable, so Bounces was used as the bounce metric."
} else {
clean_data <- clean_data %>%
  mutate(BounceMetric = BouncePerVisit)
bounce_metric_note <- "Bounces appears to be a count, so BouncePerVisit was used as the bounce metric."
}

cat(bounce_metric_note, "\n")

# Confirm cleaned data
cat("Original rows:", nrow(raw_data), "\n")
cat("Rows after cleaning:", nrow(clean_data), "\n")
cat("Rows removed:", nrow(raw_data) - nrow(clean_data), "\n\n")
